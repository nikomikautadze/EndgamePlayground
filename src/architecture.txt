src/
├── core/                        # DOMINIO (Reglas puras, sin React ni UI)
│   ├── types/                   # BoardPosition, Move, PgnGame, MinigameState
│   ├── pgn/                     # Parseo de PGN y extracción de FENs
│   └── endgame/                 # Algoritmo de detección de finales (heurísticas)
│
├── infrastructure/              # ADAPTADORES EXTERNOS
│   ├── engine/                  # Web Worker para Stockfish.js (postMessage, UCI commands)
│   ├── api/                     # Cliente para Lichess / Chess.com APIs (búsqueda de partidas)
│   └── storage/                 # Persistencia local (guardar partidas o récords del minijuego)
│
├── features/                    # MÓDULOS DE UI (React/Next.js + Custom Hooks)
│   ├── pgn-viewer/              # Componente de carga PGN, navegación por jugadas y selector
│   ├── playable-board/          # Tablero para jugar contra Stockfish o amigo desde una FEN
│   ├── evaluation-bar/          # Barra visual de ventaja según Stockfish
│   └── minigames/               # Lógica del minijuego "Adivina quién va ganando"
│
└── shared/                      # UI común (Botones, modales, componentes genéricos de layout)
